```bash
mkdir -p /home/cjj/AUV/repo_rewrite_20261117 && cd /home/cjj/AUV/repo_rewrite_20261117
cp /mnt/c/Users/cjj/Desktop/IDP3/zhidao/2026-11-17_历史重写_被删路径清单_换行分隔.txt list.txt
sha256sum list.txt   # 应为 a8e7d8b41a146220029a382670e636c4755ceba489a5cf4dc39aebf758aa2e4f

#    把 bundle 复制到 Windows 备份盘，确认复制后 sha256 一致，再继续

git clone --mirror https://github.com/cjjandtsr/diffusion_policy_in_AUV.git work.git
cd work.git
git for-each-ref --format='%(refname) %(objectname)' refs/heads > ../refs_before.txt
for b in $(git for-each-ref --format='%(refname:short)' refs/heads); do echo "$b $(git rev-parse "$b^{tree}")"; done > ../trees_before.txt
git count-objects -vH > ../size_before.txt
python3 -m pip install --user git-filter-repo     # 已装可跳过
~/.local/bin/git-filter-repo --invert-paths --paths-from-file ../list.txt; echo "filter_rc=$?"

# 3. 核对（全部通过再推送）
for b in $(git for-each-ref --format='%(refname:short)' refs/heads); do echo "$b $(git rev-parse "$b^{tree}")"; done > ../trees_after.txt
grep '^main ' ../trees_before.txt; grep '^main ' ../trees_after.txt      # 两行树哈希必须相同
git log --all --format= --name-only | sort -u > ../paths_after.txt
sort -u ../list.txt | comm -12 - ../paths_after.txt | wc -l               # 必须为 0
git count-objects -vH > ../size_after.txt; cat ../size_before.txt ../size_after.txt
diff ../trees_before.txt ../trees_after.txt                               # 记录哪些分支末端变了

# 4. 推送（不可逆）
git remote add origin https://github.com/cjjandtsr/diffusion_policy_in_AUV.git
git push --force --mirror origin; echo "push_rc=$?"
```
